57  final class CentralDogmaSotwConfigSourceSubscriptionFactory
58          implements SotwConfigSourceSubscriptionFactory {
59  
60      static final String NAME = "centraldogma.config_source";
61      static final String TYPE_URL =
62              "type.googleapis.com/com.linecorp.centraldogma.xds.v1.CentralDogmaConfigSource";
63  
64      @Override
65      public String name() {
66          return NAME;
67      }
68  
69      @Override
70      public List<String> typeUrls() {
71          return ImmutableList.of(TYPE_URL);
72      }
73  
74      @Override
75      public SnapshotStream<DiscoveryResponse> create(ConfigSource configSource,
76                                                      FactoryContext factoryContext,
77                                                      SnapshotStream<InterestedResources> interestedResources) {
78          final CentralDogmaConfigSource cdConfig =
79                  factoryContext.validator().unpack(
80                          configSource.getCustomConfigSource().getTypedConfig(),
81                          CentralDogmaConfigSource.class);
82          checkArgument(!cdConfig.getClusterName().isEmpty(),
83                        "CentralDogmaConfigSource.cluster_name must not be empty");
84          final SnapshotStream<ClusterSnapshot> clusterStream =
85                  factoryContext.clusterStream(cdConfig.getClusterName());
86          final EventExecutor eventLoop = factoryContext.eventLoop();
87          final SnapshotStream<String> accessTokenStream;
88          if (cdConfig.hasBearerTokenCredential()) {
89              final SdsSecretConfig tokenSecret = cdConfig.getBearerTokenCredential().getTokenSecret();
90              accessTokenStream = factoryContext.genericSecretStream(tokenSecret)
91                                                .map(GenericSecretSnapshot::credential);
92          } else {
93              accessTokenStream = SnapshotStream.just(CsrfToken.ANONYMOUS);
94          }
95          final SnapshotStream<CentralDogma> cdStream =
96                  SnapshotStream.combineLatest(clusterStream, accessTokenStream, Map::entry)
97                                .switchMapEager(entry -> {
98                                    return new CentralDogmaClientStream(entry.getKey(), entry.getValue(),
99                                                                        factoryContext);
100                               });
101         return cdStream.switchMapEager(centralDogma -> {
102             final Map<XdsType, InterestedResources> accumulated = new EnumMap<>(XdsType.class);
103             final Function<String, SnapshotStream<JsonNode>> watcherCache =
104                     SnapshotStream.caching(
105                             name -> new WatcherStream(centralDogma, ResourcePath.parse(name),
106                                                       eventLoop));
107             return interestedResources
108                     .map(interest -> {
109                         accumulated.put(interest.type(), interest);
110                         return ImmutableMap.copyOf(accumulated);
111                     })
112                     .switchMapEager(interests ->
113                                             new Interest2ResponsesStream(interests, watcherCache));
114         });
115     }
116 
117     private static final class Interest2ResponsesStream extends RefCountedStream<DiscoveryResponse> {
118 
119         private final Map<XdsType, InterestedResources> interests;
120         private final Function<String, SnapshotStream<JsonNode>> watcherCache;
121 
122         Interest2ResponsesStream(Map<XdsType, InterestedResources> interests,
123                                  Function<String, SnapshotStream<JsonNode>> watcherCache) {
124             this.interests = interests;
125             this.watcherCache = watcherCache;
126         }
127 
128         @Override
129         protected Subscription onStart(SnapshotWatcher<DiscoveryResponse> watcher) {
130             final List<Subscription> subs = new ArrayList<>();
131             for (InterestedResources interested : interests.values()) {
132                 final XdsType type = interested.type();
133                 final List<SnapshotStream<Any>> streams =
134                         interested.resourceNames().stream()
135                                   .map(name -> watcherCache.apply(name)
136                                                            .map(jsonNode -> toAny(jsonNode, type, name)))
137                                   .collect(ImmutableList.toImmutableList());
138                 subs.add(SnapshotStream.combineNLatest(streams)
139                                        .map(resources -> DiscoveryResponse.newBuilder()
140                                                                           .setTypeUrl(type.typeUrl())
141                                                                           .addAllResources(resources)
142                                                                           .build())
143                                        .subscribe(this::emit));
144             }
145             return () -> {
146                 subs.forEach(Subscription::close);
147                 subs.clear();
148             };
149         }
150 
151         private static Any toAny(JsonNode jsonNode, XdsType type, String resourceName) {
152             final ObjectNode copy = jsonNode.deepCopy();
153             copy.put("@type", type.typeUrl());
154             if (type == XdsType.ENDPOINT) {
155                 copy.put("cluster_name", resourceName);
156             } else {
157                 copy.put("name", resourceName);
158             }
159             return XdsResourceReader.from(copy.toString(), Any.class);
160         }
161     }

163     private static final class CentralDogmaClientStream extends RefCountedStream<CentralDogma> {
164 
165         private final ClusterSnapshot clusterSnapshot;
166         private final String accessToken;
167         private final FactoryContext factoryContext;
168 
169         CentralDogmaClientStream(ClusterSnapshot clusterSnapshot, String accessToken,
170                                  FactoryContext factoryContext) {
171             this.clusterSnapshot = clusterSnapshot;
172             this.accessToken = accessToken;
173             this.factoryContext = factoryContext;
174         }
175 
176         @Override
177         protected Subscription onStart(SnapshotWatcher<CentralDogma> watcher) {
178             final CentralDogma centralDogma = PreprocessorBasedCentralDogma.of(
179                     clusterSnapshot.preprocessor(), accessToken, factoryContext.meterRegistry());
180             emit(centralDogma, null);
181             return () -> {
182                 try {
183                     centralDogma.close();
184                 } catch (Exception e) {
185                     Exceptions.throwUnsafely(e);
186                 }
187             };
188         }
189     }
190 
191     private static final class WatcherStream extends RefCountedStream<JsonNode> {
192 
193         private final CentralDogma centralDogma;
194         private final ResourcePath resourcePath;
195         private final EventExecutor eventLoop;
196 
197         WatcherStream(CentralDogma centralDogma, ResourcePath resourcePath, EventExecutor eventLoop) {
198             this.centralDogma = centralDogma;
199             this.resourcePath = resourcePath;
200             this.eventLoop = eventLoop;
201         }
202 
203         @Override
204         protected Subscription onStart(SnapshotWatcher<JsonNode> watcher) {
205             final Watcher<?> cdWatcher;
206             if (resourcePath.isFtl()) {
207                 // .ftl files are stored as TEXT, so we use Query.ofText and parse after rendering.
208                 final WatcherRequest<String> textRequest =
209                         centralDogma.forRepo(resourcePath.project(), resourcePath.repo())
210                                     .watcher(Query.ofText(resourcePath.path()));
211                 textRequest.renderTemplate(true);
212                 if (resourcePath.profile() != null) {
213                     textRequest.renderTemplate(resourcePath.profile());
214                 }
215                 final Watcher<String> textWatcher = textRequest.start();
216                 textWatcher.watch((revision, text) -> eventLoop.execute(() -> emitText(text)));
217                 cdWatcher = textWatcher;
218             } else {
219                 final WatcherRequest<JsonNode> jsonRequest =
220                         centralDogma.forRepo(resourcePath.project(), resourcePath.repo())
221                                     .watcher(resourcePath.query());
222                 final Watcher<JsonNode> jsonWatcher = jsonRequest.start();
223                 jsonWatcher.watch((revision, jsonNode) ->
224                                           eventLoop.execute(() -> emit(jsonNode, null)));
225                 cdWatcher = jsonWatcher;
226             }
227             return cdWatcher::close;
228         }
229 
230         private void emitText(String text) {
231             try {
232                 emit(Jackson.readTree(resourcePath.basePath(), text), null);
233             } catch (Exception e) {
234                 emit(null, e);
235             }
236         }
237     }
