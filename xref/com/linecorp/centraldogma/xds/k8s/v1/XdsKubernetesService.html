124     /**
125      * POST /xds/groups/{group}/k8s/endpointAggregators
126      *
127      * <p>Creates a new Kubernetes endpoint aggregator.
128      */
129     @Post("/xds/groups/{group}/k8s/endpointAggregators")
130     @Consumes("application/yaml")
131     @RequiresXdsGroupRole(RepositoryRole.WRITE)
132     public CompletableFuture<HttpResponse> createKubernetesEndpointAggregator(
133             @Param("group") String group,
134             @Param("aggregator_id") String aggregatorId,
135             @Param("summary") @Nullable String summary,
136             String body) {
137         if (!RESOURCE_ID_PATTERN.matcher(aggregatorId).matches()) {
138             return CompletableFuture.completedFuture(
139                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
140                                                      "Invalid aggregator ID: " + aggregatorId));
141         }
142         final String kubernetesEndpointName =
143                 "groups/" + group + K8S_ENDPOINT_AGGREGATORS_DIRECTORY + aggregatorId;
144         final String clusterName = "groups/" + group + "/k8s/clusters/" + aggregatorId;
145         final KubernetesEndpointAggregator aggregator;
146         try {
147             aggregator = XdsResourceManager.parseYaml(body, KubernetesEndpointAggregator.newBuilder());
148         } catch (IOException e) {
149             return CompletableFuture.completedFuture(
150                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
151                                                      "Invalid request body: " + e.getMessage()));
152         }
153         final List<KubernetesLocalityLbEndpoints> kubernetesLocalityLbEndpointsList =
154                 aggregator.getLocalityLbEndpointsList();
155         if (kubernetesLocalityLbEndpointsList.isEmpty()) {
156             return CompletableFuture.completedFuture(
157                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
158                                                      "locality_lb_endpoints must not be empty"));
159         }
160         final Author author = currentAuthor();
161         final String aggregatorFileName = K8S_ENDPOINT_AGGREGATORS_DIRECTORY + aggregatorId + ".yaml";
162         final String createSummary = isNullOrEmpty(summary) ?
163                                      "Create kubernetes endpoint: " + kubernetesEndpointName : summary;
164         String bodyToStore = XdsResourceManager.injectYamlField(body, "name", kubernetesEndpointName);
165         bodyToStore = XdsResourceManager.injectYamlField(bodyToStore, "clusterName", clusterName);
166         final String finalBodyToStore = bodyToStore;
167         return validateKubernetesEndpointAndPushHttp(
168                 kubernetesLocalityLbEndpointsList, aggregator.hasPolicy() ? aggregator.getPolicy() : null,
169                 group, aggregatorFileName,
170                 () -> xdsResourceManager.push(group, kubernetesEndpointName,
171                                               aggregatorFileName, createSummary, author,
172                                               true, finalBodyToStore));
173     }

175     /**
176      * PUT /xds/groups/{group}/k8s/endpointAggregators/{aggregator_id}?revision={baseRevision}
177      *
178      * <p>Updates an existing Kubernetes endpoint aggregator. {@code revision} is the absolute revision the
179      * client read the aggregator at; the update is rejected with {@code 409 Conflict} if the group
180      * repository has advanced since, so a stale save cannot silently roll back a concurrent change. It
181      * defaults to {@code -1} (HEAD), which always applies.
182      */
183     @Blocking
184     @Put("/xds/groups/{group}/k8s/endpointAggregators/{*aggregator_id}")
185     @Consumes("application/yaml")
186     @RequiresXdsGroupRole(RepositoryRole.WRITE)
187     public CompletableFuture<HttpResponse> updateKubernetesEndpointAggregator(
188             @Param("group") String group,
189             @Param("aggregator_id") String aggregatorId,
190             @Param("summary") @Nullable String summary,
191             @Param("revision") @Default("-1") Revision baseRevision,
192             String body) {
193         final String aggregatorName = "groups/" + group + K8S_ENDPOINT_AGGREGATORS_DIRECTORY + aggregatorId;
194         final Matcher matcher = K8S_ENDPOINT_AGGREGATORS_NAME_PATTERN.matcher(aggregatorName);
195         if (!matcher.matches()) {
196             return CompletableFuture.completedFuture(
197                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
198                                                      "Invalid aggregator name: " + aggregatorName));
199         }
200         final String clusterName = AGGREGATORS_REPLCACE_PATTERN.matcher(aggregatorName)
201                                                                .replaceFirst("/clusters/");
202         final KubernetesEndpointAggregator aggregator;
203         try {
204             aggregator = XdsResourceManager.parseYaml(body, KubernetesEndpointAggregator.newBuilder());
205         } catch (IOException e) {
206             return CompletableFuture.completedFuture(
207                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
208                                                      "Invalid request body: " + e.getMessage()));
209         }
210         final List<KubernetesLocalityLbEndpoints> kubernetesLocalityLbEndpointsList =
211                 aggregator.getLocalityLbEndpointsList();
212         if (kubernetesLocalityLbEndpointsList.isEmpty()) {
213             return CompletableFuture.completedFuture(
214                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
215                                                      "locality_lb_endpoints must not be empty"));
216         }
217         final Author author = currentAuthor();
218         final String updateSummary = isNullOrEmpty(summary) ?
219                                      "Update kubernetes endpoint aggregator: " + aggregatorName : summary;
220         // Both name and clusterName are server-derived; inject them so the stored YAML is always correct
221         // regardless of what the client sent in the body.
222         String bodyToStore = XdsResourceManager.injectYamlField(body, "name", aggregatorName);
223         bodyToStore = XdsResourceManager.injectYamlField(bodyToStore, "clusterName", clusterName);
224         final String finalBodyToStore = bodyToStore;
225         return validateKubernetesEndpointAndPushHttp(
226                 kubernetesLocalityLbEndpointsList, aggregator.hasPolicy() ? aggregator.getPolicy() : null,
227                 group, fileName(group, aggregatorName),
228                 () -> xdsResourceManager.update(group, aggregatorName, updateSummary, author,
229                                                 finalBodyToStore, baseRevision));
230     }

254     /**
255      * POST /xds/groups/{group}/k8s/endpointAggregators:preview
256      *
257      * <p>Previews the endpoints that would be resolved for a Kubernetes endpoint aggregator
258      * without persisting it.
259      */
260     @Post("/xds/groups/{group}/k8s/endpointAggregators:preview")
261     @Consumes("application/yaml")
262     @RequiresXdsGroupRole(RepositoryRole.WRITE)
263     public CompletableFuture<HttpResponse> previewKubernetesEndpointAggregator(
264             @Param("group") String group,
265             String body) {
266         final KubernetesEndpointAggregator aggregator;
267         try {
268             aggregator = XdsResourceManager.parseYaml(
269                     body, KubernetesEndpointAggregator.newBuilder());
270         } catch (IOException e) {
271             return CompletableFuture.completedFuture(
272                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
273                                                      "Invalid request body: " + e.getMessage()));
274         }
275         final List<KubernetesLocalityLbEndpoints> localityLbEndpointsList =
276                 aggregator.getLocalityLbEndpointsList();
277         if (localityLbEndpointsList.isEmpty()) {
278             return CompletableFuture.completedFuture(
279                     XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST,
280                                                      "locality_lb_endpoints must not be empty"));
281         }
282         final MetaRepository metaRepository = xdsResourceManager.xdsProject().metaRepo();
283         final ContextAwareBlockingTaskExecutor taskExecutor =
284                 ServiceRequestContext.current().blockingTaskExecutor();
285 
286         final List<CompletableFuture<LocalityLbEndpoints>> futures = new ArrayList<>();
287         for (KubernetesLocalityLbEndpoints localityLbEndpoints : localityLbEndpointsList) {
288             futures.add(resolvePreview(localityLbEndpoints, group, metaRepository, taskExecutor));
289         }
290 
291         return CompletableFuture.allOf(futures.toArray(EMPTY_FUTURES))
292                                 .handle((unused, cause) -> {
293                                     if (cause != null) {
294                                         return toErrorHttpResponse(cause);
295                                     }
296                                     final ClusterLoadAssignment.Builder cla =
297                                             ClusterLoadAssignment.newBuilder();
298                                     if (!aggregator.getClusterName().isEmpty()) {
299                                         cla.setClusterName(aggregator.getClusterName());
300                                     }
301                                     if (aggregator.hasPolicy()) {
302                                         cla.setPolicy(aggregator.getPolicy());
303                                     }
304                                     for (CompletableFuture<LocalityLbEndpoints> future : futures) {
305                                         cla.addEndpoints(future.join());
306                                     }
307                                     try {
308                                         return XdsResourceManager.toYamlResponse(cla.build());
309                                     } catch (IOException e) {
310                                         return XdsResourceManager.errorResponse(
311                                                 HttpStatus.INTERNAL_SERVER_ERROR, e);
312                                     }
313                                 });
314     }

316     private CompletableFuture<HttpResponse> validateKubernetesEndpointAndPushHttp(
317             List<KubernetesLocalityLbEndpoints> kubernetesLocalityLbEndpointsList,
318             @Nullable Policy policy, String group, String fileNameForLookup,
319             Supplier<CompletableFuture<HttpResponse>> onSuccess) {
320         for (KubernetesLocalityLbEndpoints kubernetesLocalityLbEndpoints : kubernetesLocalityLbEndpointsList) {
321             try {
322                 validateMetadataMappings(kubernetesLocalityLbEndpoints.getWatcher());
323             } catch (IllegalArgumentException e) {
324                 return CompletableFuture.completedFuture(
325                         XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST, e));
326             }
327         }
328         if (policy != null) {
329             try {
330                 validatePolicy(policy);
331             } catch (IllegalArgumentException e) {
332                 return CompletableFuture.completedFuture(
333                         XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST, e));
334             }
335         }
336         final ContextAwareBlockingTaskExecutor taskExecutor =
337                 ServiceRequestContext.current().blockingTaskExecutor();
338 
339         final ArrayList<CompletableFuture<Void>> futures = new ArrayList<>();
340         for (KubernetesLocalityLbEndpoints kubernetesLocalityLbEndpoints : kubernetesLocalityLbEndpointsList) {
341             final CompletableFuture<Void> future = new CompletableFuture<>();
342             futures.add(future);
343 
344             final ServiceEndpointWatcher watcher = kubernetesLocalityLbEndpoints.getWatcher();
345             final CompletableFuture<KubernetesEndpointGroup> endpointGroupFuture =
346                     createKubernetesEndpointGroup(watcher, xdsResourceManager.xdsProject().metaRepo(),
347                                                   group, fileNameForLookup, false);
348             endpointGroupFuture.handle((kubernetesEndpointGroup, cause) -> {
349                 if (cause != null) {
350                     future.completeExceptionally(Exceptions.peel(cause));
351                     return null;
352                 }
353                 final AtomicBoolean completed = new AtomicBoolean();
354                 final CompletableFuture<List<Endpoint>> whenReady = kubernetesEndpointGroup.whenReady();
355                 final ScheduledFuture<?> scheduledFuture = taskExecutor.schedule(() -> {
356                     if (!completed.compareAndSet(false, true)) {
357                         return;
358                     }
359                     kubernetesEndpointGroup.closeAsync();
360                     future.completeExceptionally(new IllegalStateException(
361                             "Failed to retrieve k8s endpoints within 5 seconds. watcher: " + watcher));
362                 }, 5, TimeUnit.SECONDS);
363 
364                 whenReady.handle((endpoints, cause1) -> {
365                     if (!completed.compareAndSet(false, true)) {
366                         return null;
367                     }
368                     scheduledFuture.cancel(false);
369                     kubernetesEndpointGroup.closeAsync();
370                     if (cause1 != null) {
371                         future.completeExceptionally(
372                                 new IllegalStateException("Failed to retrieve k8s endpoints", cause1));
373                     } else {
374                         logger.debug("Successfully retrieved k8s endpoints: {}, watcher: {}",
375                                      endpoints, watcher);
376                         future.complete(null);
377                     }
378                     return null;
379                 });
380                 return null;
381             });
382         }
383 
384         return CompletableFuture.allOf(futures.toArray(EMPTY_FUTURES))
385                                 .<HttpResponse>handle((unused, cause) -> {
386                                     if (cause != null) {
387                                         return toErrorHttpResponse(cause);
388                                     }
389                                     return null;
390                                 })
391                                 .thenCompose(errorResponse -> {
392                                     if (errorResponse != null) {
393                                         return CompletableFuture.completedFuture(errorResponse);
394                                     }
395                                     return onSuccess.get();
396                                 });
397     }

399     private static HttpResponse toErrorHttpResponse(Throwable cause) {
400         final Throwable peeled = Exceptions.peel(cause);
401         // invalid credential not found
402         if (peeled instanceof IllegalArgumentException || peeled instanceof EntryNotFoundException) {
403             return XdsResourceManager.errorResponse(HttpStatus.BAD_REQUEST, peeled);
404         }
405 
406         return XdsResourceManager.errorResponse(HttpStatus.INTERNAL_SERVER_ERROR, peeled);
407     }

409     private static CompletableFuture<LocalityLbEndpoints> resolvePreview(
410             KubernetesLocalityLbEndpoints localityLbEndpoints, String group,
411             MetaRepository metaRepository, ContextAwareBlockingTaskExecutor taskExecutor) {
412         final CompletableFuture<LocalityLbEndpoints> result = new CompletableFuture<>();
413         final ServiceEndpointWatcher watcher = localityLbEndpoints.getWatcher();
414         createKubernetesEndpointGroup(watcher, metaRepository, group, "(preview)", false)
415                 .handle((endpointGroup, cause) -> {
416                     if (cause != null) {
417                         result.completeExceptionally(Exceptions.peel(cause));
418                         return null;
419                     }
420                     final AtomicBoolean completed = new AtomicBoolean();
421                     final ScheduledFuture<?> timeout = taskExecutor.schedule(() -> {
422                         if (completed.compareAndSet(false, true)) {
423                             endpointGroup.closeAsync();
424                             result.completeExceptionally(new IllegalStateException(
425                                     "Timed out after " + PREVIEW_TIMEOUT_SECONDS +
426                                     "s resolving '" + watcher.getServiceName() + '\''));
427                         }
428                     }, PREVIEW_TIMEOUT_SECONDS, TimeUnit.SECONDS);
429                     endpointGroup.whenReady().handle((endpoints, readyCause) -> {
430                         if (!completed.compareAndSet(false, true)) {
431                             return null;
432                         }
433                         timeout.cancel(false);
434                         try {
435                             if (readyCause != null) {
436                                 result.completeExceptionally(Exceptions.peel(readyCause));
437                             } else {
438                                 result.complete(toLocalityLbEndpoints(endpointGroup, localityLbEndpoints));
439                             }
440                         } finally {
441                             endpointGroup.closeAsync();
442                         }
443                         return null;
444                     });
445                     return null;
446                 });
447         return result;
448     }

450     private static LocalityLbEndpoints toLocalityLbEndpoints(
451             KubernetesEndpointGroup endpointGroup, KubernetesLocalityLbEndpoints localityLbEndpoints) {
452         final LocalityLbEndpoints.Builder builder = LocalityLbEndpoints.newBuilder();
453         if (localityLbEndpoints.hasLocality()) {
454             builder.setLocality(localityLbEndpoints.getLocality());
455         }
456         if (localityLbEndpoints.hasLoadBalancingWeight()) {
457             builder.setLoadBalancingWeight(localityLbEndpoints.getLoadBalancingWeight());
458         }
459         builder.setPriority(localityLbEndpoints.getPriority());
460         KubernetesEndpointConverter.addLbEndpoints(builder, endpointGroup.endpoints(),
461                                                    localityLbEndpoints.getWatcher());
462         return builder.build();
463     }

465     // The policy is copied into the generated ClusterLoadAssignment, so a value Envoy rejects would take
466     // the whole endpoint set down with it.
467     private static void validatePolicy(Policy policy) {
468         if (policy.getDropOverloadsCount() > 1) {
469             throw new IllegalArgumentException(
470                     "at most one drop_overload is supported, but got: " + policy.getDropOverloadsCount());
471         }
472         for (DropOverload dropOverload : policy.getDropOverloadsList()) {
473             if (dropOverload.getCategory().isEmpty()) {
474                 throw new IllegalArgumentException("category must not be empty in drop_overloads");
475             }
476         }
477         if (policy.hasOverprovisioningFactor() && policy.getOverprovisioningFactor().getValue() == 0) {
478             throw new IllegalArgumentException("overprovisioning_factor must be greater than 0");
479         }
480     }

482     private static void validateMetadataMappings(ServiceEndpointWatcher watcher) {
483         for (MetadataMapping mapping : watcher.getMetadataMappingList()) {
484             if (mapping.getResourceType() == MetadataMapping.ResourceType.RESOURCE_TYPE_UNSPECIFIED) {
485                 throw new IllegalArgumentException(
486                         "resource_type must be specified in metadata_mapping: " + mapping);
487             }
488             if (mapping.getEntryType() == MetadataMapping.EntryType.ENTRY_TYPE_UNSPECIFIED) {
489                 throw new IllegalArgumentException(
490                         "entry_type must be specified in metadata_mapping: " + mapping);
491             }
492             switch (mapping.getSourceCase()) {
493                 case SOURCE_KEY:
494                     if (mapping.getSourceKey().isEmpty()) {
495                         throw new IllegalArgumentException(
496                                 "source_key must not be empty in metadata_mapping: " + mapping);
497                     }
498                     break;
499                 case SOURCE_KEY_PREFIX:
500                     if (mapping.getSourceKeyPrefix().isEmpty()) {
501                         throw new IllegalArgumentException(
502                                 "source_key_prefix must not be empty in metadata_mapping: " + mapping);
503                     }
504                     break;
505                 default:
506                     throw new IllegalArgumentException(
507                             "either source_key or source_key_prefix must be set in metadata_mapping: " +
508                             mapping);
509             }
510         }
511     }

513     /**
514      * Creates a {@link KubernetesEndpointGroup} from the specified {@link ServiceEndpointWatcher}.
515      */
516     public static CompletableFuture<KubernetesEndpointGroup> createKubernetesEndpointGroup(
517             ServiceEndpointWatcher watcher, MetaRepository metaRepository, String group, String fileName,
518             boolean logIfFail) {
519         final Kubeconfig kubeconfig = watcher.getKubeconfig();
520         final String serviceName = watcher.getServiceName();
521 
522         final CompletableFuture<KubernetesEndpointGroup> future =
523                 toConfig(kubeconfig, metaRepository, group, fileName).thenApply(config -> {
524                     final KubernetesEndpointGroupBuilder kubernetesEndpointGroupBuilder =
525                             KubernetesEndpointGroup.builder(config).serviceName(serviceName);
526                     if (!isNullOrEmpty(kubeconfig.getNamespace())) {
527                         kubernetesEndpointGroupBuilder.namespace(kubeconfig.getNamespace());
528                     }
529                     if (!isNullOrEmpty(watcher.getPortName())) {
530                         kubernetesEndpointGroupBuilder.portName(watcher.getPortName());
531                     }
532                     if (nodeIpExtractor != null) {
533                         kubernetesEndpointGroupBuilder.nodeIpExtractor(
534                                 node -> nodeIpExtractor.extract(watcher, node));
535                     }
536                     return kubernetesEndpointGroupBuilder.build();
537                 });
538         if (logIfFail) {
539             future.exceptionally(cause -> {
540                 logger.warn("Failed to create KubernetesEndpointGroup. watcher: {}", watcher, cause);
541                 return null;
542             });
543         }
544 
545         return future;
546     }

548     private static CompletableFuture<Config> toConfig(Kubeconfig kubeconfig, MetaRepository metaRepository,
549                                                       String group, String fileName) {
550         final ConfigBuilder configBuilder = new ConfigBuilder()
551                 .withMasterUrl(kubeconfig.getControlPlaneUrl())
552                 .withTrustCerts(kubeconfig.getTrustCerts());
553 
554         final String credentialId = kubeconfig.getCredentialId();
555         if (isNullOrEmpty(credentialId)) {
556             return CompletableFuture.completedFuture(configBuilder.build());
557         }
558         final CompletableFuture<Config> future = new CompletableFuture<>();
559         // xDS only support repository credential so try using it first.
560         metaRepository.credential(credentialName(INTERNAL_PROJECT_XDS, group, credentialId))
561                       .thenAccept(credential -> {
562                           if (!(credential instanceof AccessTokenCredential)) {
563                               future.completeExceptionally(new IllegalArgumentException(
564                                       "credential must be an access token: " + credential.withoutSecret()));
565                           } else {
566                               future.complete(configBuilder.withOauthToken(
567                                       ((AccessTokenCredential) credential).accessToken()).build());
568                           }
569                       })
570                       .exceptionally(cause -> {
571                           final Throwable peeled = Exceptions.peel(cause);
572                           if (peeled instanceof EntryNotFoundException) {
573                               // Try to use the legacy project credential for backward compatibility.
574                               metaRepository.credential(credentialName(INTERNAL_PROJECT_XDS, credentialId))
575                                             .handle((credential, cause1) -> {
576                                                 if (cause1 != null) {
577                                                     future.completeExceptionally(cause1);
578                                                     return null;
579                                                 }
580                                                 if (!(credential instanceof AccessTokenCredential)) {
581                                                     future.completeExceptionally(new IllegalArgumentException(
582                                                             "credential must be an access token: " +
583                                                             credential.withoutSecret()));
584                                                 } else {
585                                                     logger.warn("Project credential is used in {}. " +
586                                                                 "Please migrate to group {}. credential ID: {}",
587                                                                 fileName, group, credentialId);
588                                                     final String token =
589                                                             ((AccessTokenCredential) credential).accessToken();
590                                                     future.complete(
591                                                             configBuilder.withOauthToken(token).build());
592                                                 }
593                                                 return null;
594                                             });
595                           } else {
596                               future.completeExceptionally(peeled);
597                           }
598                           return null;
599                       });
600         return future;
601     }
