76      public static final MessageMarshaller JSON_MESSAGE_MARSHALLER;
77  
78      // Packages of proto types that xDS resources may embed in a google.protobuf.Any (e.g. an HttpFilter
79      // typed_config) but that do not live under io.envoyproxy.envoy, so the Envoy scan does not pick them up.
80      // These are the Armeria xDS filter configs and their metadata messages, provided by armeria-xds-api.
81      private static final ImmutableList<String> CUSTOM_TYPE_PACKAGES = ImmutableList.of(
82              "com.linecorp.armeria.xds.athenz",
83              "jp.co.lycorp.ftd.athenz.v1");
84  
85      static {
86          final MessageMarshaller.Builder builder =
87                  MessageMarshaller.builder().omittingInsignificantWhitespace(true);
88          // These Central Dogma-specific types are not in the io.envoyproxy.envoy package, so they
89          // must be registered explicitly; the io.envoyproxy.envoy scan does not pick them up.
90          builder.register(KubernetesEndpointAggregator.getDefaultInstance())
91                 .register(LocalityLbEndpoint.getDefaultInstance());
92          // Envoy types use java_multiple_files, so every message is a top-level class.
93          registerPackage(builder, "io.envoyproxy.envoy", false);
94          // The custom (non-Envoy) protos do not set java_multiple_files, so their messages are nested.
95          for (String customTypePackage : CUSTOM_TYPE_PACKAGES) {
96              registerPackage(builder, customTypePackage, true);
97          }
98          JSON_MESSAGE_MARSHALLER = builder.build();
99      }
100 
101     private static void registerPackage(MessageMarshaller.Builder builder, String packageName,
102                                         boolean includeNested) {
103         final Reflections reflections = new Reflections(
104                 packageName, XdsResourceManager.class.getClassLoader(),
105                 new SubTypesScanner(true));
106         reflections.getSubTypesOf(GeneratedMessageV3.class)
107                    .stream()
108                    // For java_multiple_files packages, a nested ('$') class is never a registrable message.
109                    // For custom packages the messages themselves are nested, so nested classes are kept and
110                    // the non-message ones (Builders, enums) are filtered out below by getDefaultInstance check.
111                    .filter(c -> includeNested || !c.getName().contains("$"))
112                    .filter(XdsResourceManager::hasGetDefaultInstanceMethod)
113                    .forEach(c -> {
114                        // register() does not throw; build() does. A test build per class is needed
115                        // to detect unsupported types before adding them to the real builder.
116                        try {
117                            MessageMarshaller.builder()
118                                             .omittingInsignificantWhitespace(true)
119                                             .register(c)
120                                             .build();
121                            builder.register(c);
122                        } catch (Exception e) {
123                            logger.debug("Skipping proto type not supported by MessageMarshaller: {}",
124                                         c.getName());
125                        }
126                    });
127     }
128 
129     private static boolean hasGetDefaultInstanceMethod(Class<?> clazz) {
130         try {
131             final Method method = clazz.getMethod("getDefaultInstance");
132             return method.getParameterCount() == 0 && Modifier.isStatic(method.getModifiers());
133         } catch (NoSuchMethodException ignored) {
134             return false;
135         }
136     }

179     public static String toYamlBodyString(Message resource) throws IOException {
180         final String json = JSON_MESSAGE_MARSHALLER.writeValueAsString(resource);
181         final JsonNode node = Jackson.readTree(json);
182         return Yaml.writeValueAsString(node);
183     }

185     @SuppressWarnings("unchecked")
186     public static <T extends Message> T parseYaml(String body, Message.Builder builder) throws IOException {
187         JSON_MESSAGE_MARSHALLER.mergeValue(Yaml.readTree(body).traverse(), builder);
188         return (T) builder.build();
189     }

191     /**
192      * Injects or replaces a top-level YAML field in the given YAML string, preserving all other
193      * content including comments. If {@code fieldName} is not found as a top-level key, it is
194      * prepended.
195      */
196     public static String injectYamlField(String yaml, String fieldName, String fieldValue) {
197         final String replacement = fieldName + ": " + fieldValue + '\n';
198         // Match a top-level key (no leading whitespace) in either camelCase or snake_case form,
199         // plus its entire value — including any indented block-scalar continuation lines.
200         // (?:\r?\n|\z) handles LF, CRLF, and a missing trailing newline at end of file.
201         final Pattern pattern = Pattern.compile(
202                 "^(?:" + Pattern.quote(fieldName) + '|' +
203                 Pattern.quote(camelToSnake(fieldName)) +
204                 ")\\s*:.*(?:\\r?\\n|\\z)(?:[ \\t]+.*(?:\\r?\\n|\\z))*",
205                 Pattern.MULTILINE);
206         final Matcher matcher = pattern.matcher(yaml);
207         if (matcher.find()) {
208             return matcher.replaceFirst(Matcher.quoteReplacement(replacement));
209         }
210         return replacement + yaml;
211     }
212 
213     private static String camelToSnake(String camel) {
214         return camel.replaceAll("([A-Z])", "_$1").toLowerCase();
215     }

217     public CompletableFuture<HttpResponse> push(
218             String group, String resourceName, String fileName, String summary,
219             Author author, boolean create, String originalBody) {
220         if (create) {
221             final Repository repository;
222             try {
223                 repository = xdsProject.repos().get(group);
224             } catch (Exception e) {
225                 return CompletableFuture.completedFuture(
226                         errorResponse(HttpStatus.NOT_FOUND, "Group not found: " + group));
227             }
228             // Note: There is a TOCTOU race between this check and the subsequent ofYamlUpsert —
229             // two concurrent creates can both pass this check and the second will silently overwrite
230             // the first. This is acceptable for xDS resources where concurrent creates of the same
231             // resource are extremely rare.
232             return repository.find(Revision.HEAD, fileName, FIND_ONE_WITHOUT_CONTENT)
233                              .thenCompose(entries -> {
234                                  if (!entries.isEmpty()) {
235                                      return CompletableFuture.completedFuture(
236                                              errorResponse(HttpStatus.CONFLICT,
237                                                            "Resource already exists: " + resourceName));
238                                  }
239                                  return doPush(group, fileName, summary, author, true, originalBody);
240                              });
241         }
242         return doPush(group, fileName, summary, author, false, originalBody);
243     }
244 
245     private CompletableFuture<HttpResponse> doPush(
246             String group, String fileName, String summary,
247             Author author, boolean create, String originalBody) {
248         return doPush(group, fileName, summary, author, create, originalBody,
249                       Revision.HEAD);
250     }
251 
252     private CompletableFuture<HttpResponse> doPush(
253             String group, String fileName, String summary,
254             Author author, boolean create, String originalBody, Revision baseRevision) {
255         // Store the original YAML body as-is (server-set fields are injected by the caller before
256         // this method is invoked). Respond with the same body so the client sees exactly what is stored.
257         final Change<?> change = Change.ofYamlUpsert(fileName, originalBody);
258         return commandExecutor.execute(Command.push(author, INTERNAL_PROJECT_XDS, group, baseRevision,
259                                                     summary, "", Markup.PLAINTEXT,
260                                                     ImmutableList.of(change)))
261                               .handle((unused, cause) -> {
262                                   if (cause != null) {
263                                       final Throwable peeled = Exceptions.peel(cause);
264                                       if (!create && peeled instanceof RedundantChangeException) {
265                                           return toYamlResponse(originalBody);
266                                       }
267                                       if (peeled instanceof ChangeConflictException) {
268                                           return errorResponse(HttpStatus.CONFLICT, peeled);
269                                       }
270                                       if (peeled instanceof RevisionNotFoundException) {
271                                           return errorResponse(HttpStatus.BAD_REQUEST,
272                                                                "Invalid revision: " + baseRevision);
273                                       }
274                                       return errorResponse(HttpStatus.INTERNAL_SERVER_ERROR, peeled);
275                                   }
276                                   return toYamlResponse(originalBody);
277                               });
278     }

285     // A base revision other than HEAD makes the commit a compare-and-swap: it is rejected with 409 once the
286     // group repository has advanced past it, so a save based on a stale read cannot overwrite a newer one.
287     public CompletableFuture<HttpResponse> update(
288             String group, String resourceName, String summary, Author author, String originalBody,
289             Revision baseRevision) {
290         return update(group, resourceName, fileName(group, resourceName), summary, author, originalBody,
291                       baseRevision);
292     }

294     private CompletableFuture<HttpResponse> update(
295             String group, String resourceName, String fileName, String summary,
296             Author author, String originalBody, Revision baseRevision) {
297         return updateOrDelete(group, resourceName, fileName,
298                               () -> doPush(group, fileName, summary, author,
299                                            false, originalBody, baseRevision));
300     }

307     public CompletableFuture<HttpResponse> delete(
308             String group, String resourceName, String fileName, String summary, Author author) {
309         return updateOrDelete(group, resourceName, fileName, () ->
310                 commandExecutor.execute(Command.push(author, INTERNAL_PROJECT_XDS, group,
311                                                      Revision.HEAD, summary, "", Markup.PLAINTEXT,
312                                                      ImmutableList.of(Change.ofRemoval(fileName))))
313                                .handle((unused, cause) -> {
314                                    if (cause != null) {
315                                        return errorResponse(HttpStatus.INTERNAL_SERVER_ERROR,
316                                                             Exceptions.peel(cause));
317                                    }
318                                    return HttpResponse.of(HttpStatus.NO_CONTENT);
319                                }));
320     }
321 
322     public CompletableFuture<HttpResponse> updateOrDelete(
323             String group, String resourceName, String fileName,
324             Supplier<CompletableFuture<HttpResponse>> task) {
325         final Repository repository;
326         try {
327             repository = xdsProject.repos().get(group);
328         } catch (Exception e) {
329             return CompletableFuture.completedFuture(
330                     errorResponse(HttpStatus.NOT_FOUND, "Group not found: " + group));
331         }
332         return repository.find(Revision.HEAD, fileName, FIND_ONE_WITHOUT_CONTENT)
333                          .thenCompose(entries -> {
334                              if (entries.isEmpty()) {
335                                  return CompletableFuture.completedFuture(
336                                          errorResponse(HttpStatus.NOT_FOUND,
337                                                        "Resource not found: " + resourceName));
338                              }
339                              return task.get();
340                          });
341     }
