16  package com.linecorp.centraldogma.xds.internal;
17  
18  import java.util.List;
19  import java.util.Map;
20  import java.util.Set;
21  import java.util.function.Consumer;
22  
23  import org.slf4j.Logger;
24  import org.slf4j.LoggerFactory;
25  
26  import com.google.common.collect.ImmutableList;
27  import com.google.common.collect.ImmutableMap;
28  import com.google.protobuf.Message;
29  
30  import com.linecorp.armeria.common.util.SafeCloseable;
31  import com.linecorp.armeria.server.ServiceRequestContext;
32  import com.linecorp.centraldogma.xds.XdsResourceTransformer;
33  
34  import io.envoyproxy.controlplane.cache.ConfigWatcher;
35  import io.envoyproxy.controlplane.cache.DeltaResponse;
36  import io.envoyproxy.controlplane.cache.DeltaWatch;
37  import io.envoyproxy.controlplane.cache.DeltaXdsRequest;
38  import io.envoyproxy.controlplane.cache.Response;
39  import io.envoyproxy.controlplane.cache.VersionedResource;
40  import io.envoyproxy.controlplane.cache.Watch;
41  import io.envoyproxy.controlplane.cache.XdsRequest;
42  import io.envoyproxy.envoy.config.core.v3.Node;
43  import io.netty.util.AttributeKey;
44  
45  final class NodeAwareConfigWatcher implements ConfigWatcher {
46  
47      private static final Logger logger = LoggerFactory.getLogger(NodeAwareConfigWatcher.class);
48  
49      private static final AttributeKey<Node> NODE_KEY = AttributeKey.valueOf(
50              NodeAwareConfigWatcher.class, "xds.node");
51  
52      private final ConfigWatcher delegate;
53      private final List<XdsResourceTransformer> transformers;
54  
55      NodeAwareConfigWatcher(ConfigWatcher delegate, List<XdsResourceTransformer> transformers) {
56          this.delegate = delegate;
57          this.transformers = ImmutableList.copyOf(transformers);
58      }
59  
60      @Override
61      public Watch createWatch(boolean ads, XdsRequest request, Set<String> knownResourceNames,
62                               Consumer<Response> responseConsumer,
63                               boolean hasClusterChanged, boolean allowDefaultEmptyEdsUpdate) {
64          if (transformers.isEmpty()) {
65              return delegate.createWatch(ads, request, knownResourceNames,
66                                          responseConsumer, hasClusterChanged, allowDefaultEmptyEdsUpdate);
67          }
68  
69          final ServiceRequestContext ctx = ServiceRequestContext.current();
70          final Node node = resolveNode(request.v3Request().getNode(), ctx);
71          final Consumer<Response> transforming = response -> {
72              try (SafeCloseable ignored = ctx.push()) {
73                  deliverTransformed(response, node, request, responseConsumer);
74              }
75          };
76          return delegate.createWatch(ads, request, knownResourceNames,
77                                      transforming, hasClusterChanged, allowDefaultEmptyEdsUpdate);
78      }
79  
80      private void deliverTransformed(Response response, Node node, XdsRequest request,
81                                      Consumer<Response> responseConsumer) {
82          Response toDeliver;
83          try {
84              final List<Message> transformed = transformAll(node, response.resources(), request.getTypeUrl());
85              toDeliver = Response.create(request, transformed, response.version());
86          } catch (Exception e) {
87              logger.warn("Failed to transform xDS resources for node '{}'; delivering untransformed response.",
88                          node.getId(), e);
89              toDeliver = response;
90          }
91          responseConsumer.accept(toDeliver);
92      }
93  
94      @Override
95      public DeltaWatch createDeltaWatch(DeltaXdsRequest request, String requiredVersion,
96                                         Map<String, String> pendingResources,
97                                         Set<String> knownResourceNames, boolean isWildcard,
98                                         Consumer<DeltaResponse> responseConsumer,
99                                         boolean allowDefaultEmptyEdsUpdate) {
100         if (transformers.isEmpty()) {
101             return delegate.createDeltaWatch(request, requiredVersion, pendingResources,
102                                              knownResourceNames, isWildcard,
103                                              responseConsumer, allowDefaultEmptyEdsUpdate);
104         }
105 
106         final ServiceRequestContext ctx = ServiceRequestContext.current();
107         final Node node = resolveNode(request.v3Request().getNode(), ctx);
108         final Consumer<DeltaResponse> transforming = response -> {
109             try (SafeCloseable ignored = ctx.push()) {
110                 deliverDeltaTransformed(response, node, request, responseConsumer);
111             }
112         };
113         return delegate.createDeltaWatch(request, requiredVersion, pendingResources,
114                                          knownResourceNames, isWildcard,
115                                          transforming, allowDefaultEmptyEdsUpdate);
116     }
117 
118     private void deliverDeltaTransformed(DeltaResponse response, Node node, DeltaXdsRequest request,
119                                          Consumer<DeltaResponse> responseConsumer) {
120         DeltaResponse toDeliver;
121         try {
122             final String typeUrl = request.getTypeUrl();
123             final ImmutableMap.Builder<String, VersionedResource<?>> builder = ImmutableMap.builder();
124             for (Map.Entry<String, VersionedResource<?>> entry : response.resources().entrySet()) {
125                 final VersionedResource<?> vr = entry.getValue();
126                 final Message transformed = applyTransformers(node, vr.resource(), typeUrl);
127                 builder.put(entry.getKey(), VersionedResource.create(transformed, vr.version()));
128             }
129             toDeliver = DeltaResponse.create(request, builder.build(),
130                                              response.removedResources(), response.version());
131         } catch (Exception e) {
132             logger.warn("Failed to transform xDS resources for node '{}'; delivering untransformed response.",
133                         node.getId(), e);
134             toDeliver = response;
135         }
136         responseConsumer.accept(toDeliver);
137     }
138 
139     private List<Message> transformAll(Node node, Iterable<? extends Message> resources, String typeUrl) {
140         final ImmutableList.Builder<Message> result = ImmutableList.builder();
141         for (Message resource : resources) {
142             result.add(applyTransformers(node, resource, typeUrl));
143         }
144         return result.build();
145     }
146 
147     private Message applyTransformers(Node node, Message resource, String typeUrl) {
148         Message result = resource;
149         for (XdsResourceTransformer transformer : transformers) {
150             result = transformer.transform(node, result, typeUrl);
151         }
152         return result;
153     }
154 
155     /**
156      * Resolves the {@link Node} for the current stream. If a non-empty node is sent, it is stored in the
157      * {@link ServiceRequestContext} and used for subsequent requests. If an empty node is sent, the latest
158      * non-empty node is used.
159      */
160     private static Node resolveNode(Node requestNode, ServiceRequestContext ctx) {
161         if (requestNode.getSerializedSize() > 0) {
162             ctx.setAttr(NODE_KEY, requestNode);
163             return requestNode;
164         }
165         final Node stored = ctx.attr(NODE_KEY);
166         return stored != null ? stored : Node.getDefaultInstance();
167     }
168 }
