42  /**
43   * Annotated service object for managing xDS groups.
44   */
45  public final class XdsGroupService {
46  
47      private final Project xdsProject;
48      private final CommandExecutor commandExecutor;
49      private final MetadataService mds;
50  
51      /**
52       * Creates a new instance.
53       */
54      public XdsGroupService(Project xdsProject, CommandExecutor commandExecutor, MetadataService mds) {
55          this.xdsProject = xdsProject;
56          this.commandExecutor = commandExecutor;
57          this.mds = mds;
58      }
59  
60      /**
61       * POST /xds/groups
62       *
63       * <p>Creates a new xDS group.
64       */
65      @Post("/xds/groups")
66      public CompletableFuture<HttpResponse> createGroup(@Param("group_id") String groupId) {
67          if (!PROJECT_AND_REPO_NAME_PATTERN.matcher(groupId).matches()) {
68              return CompletableFuture.completedFuture(
69                      errorResponse(HttpStatus.BAD_REQUEST, "Invalid group ID: " + groupId));
70          }
71          if (Project.isInternalRepo(groupId)) {
72              return CompletableFuture.completedFuture(
73                      errorResponse(HttpStatus.FORBIDDEN, "Cannot create internal repository: " + groupId));
74          }
75          if (xdsProject.repos().exists(groupId)) {
76              return CompletableFuture.completedFuture(
77                      errorResponse(HttpStatus.CONFLICT, "Group already exists: " + groupId));
78          }
79          final User createUser = currentUser();
80          if (createUser == null) {
81              return CompletableFuture.completedFuture(
82                      errorResponse(HttpStatus.UNAUTHORIZED, "Authentication required"));
83          }
84          return createRepository(commandExecutor, mds, getAuthor(createUser), INTERNAL_PROJECT_XDS, groupId,
85                                  false, null)
86                  .handle((unused, cause) -> {
87                      if (cause != null) {
88                          final Throwable peeled = Exceptions.peel(cause);
89                          if (peeled instanceof RepositoryExistsException) {
90                              return errorResponse(HttpStatus.CONFLICT,
91                                                   "Group already exists: " + groupId);
92                          }
93                          return errorResponse(HttpStatus.INTERNAL_SERVER_ERROR, peeled);
94                      }
95                      return HttpResponse.of(HttpStatus.OK);
96                  });
97      }
98  
99      /**
100      * DELETE /xds/groups/{group_name}
101      *
102      * <p>Removes an xDS group.
103      */
104     @Delete("/xds/groups/{group_name}")
105     public CompletableFuture<HttpResponse> deleteGroup(@Param("group_name") String groupName) {
106         if (!xdsProject.repos().exists(groupName)) {
107             return CompletableFuture.completedFuture(
108                     errorResponse(HttpStatus.NOT_FOUND, "Group not found: " + groupName));
109         }
110         if (Project.isInternalRepo(groupName)) {
111             return CompletableFuture.completedFuture(
112                     errorResponse(HttpStatus.FORBIDDEN, "Cannot delete internal repository: " + groupName));
113         }
114         final User user = currentUser();
115         if (user == null) {
116             return CompletableFuture.completedFuture(
117                     errorResponse(HttpStatus.UNAUTHORIZED, "Authentication required"));
118         }
119         final ProjectMetadata metadata = xdsProject.metadata();
120         // @xds is not the internal dogma project, so metadata is always initialized — never null.
121         assert metadata != null;
122         final RepositoryRole role = MetadataService.findRepositoryRole(metadata, groupName, user);
123         if (role != RepositoryRole.ADMIN) {
124             return CompletableFuture.completedFuture(
125                     errorResponse(HttpStatus.FORBIDDEN, "No admin permission for group: " + groupName));
126         }
127         return removeRepository(commandExecutor, mds, getAuthor(user), INTERNAL_PROJECT_XDS, groupName)
128                 .handle((unused, cause) -> {
129                     if (cause != null) {
130                         return errorResponse(HttpStatus.INTERNAL_SERVER_ERROR,
131                                              Exceptions.peel(cause));
132                     }
133                     return HttpResponse.of(HttpStatus.NO_CONTENT);
134                 });
135     }
