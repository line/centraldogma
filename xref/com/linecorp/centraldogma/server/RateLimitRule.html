16  package com.linecorp.centraldogma.server;
17  
18  import static com.google.common.base.Preconditions.checkArgument;
19  import static java.util.Objects.requireNonNull;
20  
21  import java.util.concurrent.TimeUnit;
22  import java.util.regex.Pattern;
23  
24  import org.jspecify.annotations.Nullable;
25  
26  import com.fasterxml.jackson.annotation.JsonCreator;
27  import com.fasterxml.jackson.annotation.JsonInclude;
28  import com.fasterxml.jackson.annotation.JsonInclude.Include;
29  import com.fasterxml.jackson.annotation.JsonProperty;
30  import com.google.common.base.MoreObjects;
31  
32  /**
33   * A token bucket's capacity and refill rate, expressed as a request count per time unit.
34   * Buckets start full and refill gradually, up to their capacity.
35   * A count of {@code -1} allows unlimited requests, and {@code 0} rejects all requests.
36   * Global limits omit the pattern; repository rules require a pattern matched against the entire
37   * {@code projectName/repoName} and give each repository its own token bucket.
38   */
39  public final class RateLimitRule {
40  
41      @Nullable
42      private final Pattern pattern;
43      private final long requests;
44      private final TimeUnit timeUnit;
45  
46      /**
47       * Creates a global limit with capacity {@code requests}, replenishing that many tokens
48       * per {@code timeUnit}.
49       * A count of {@code -1} allows unlimited requests, and {@code 0} rejects all requests.
50       * The count must be at least {@code -1}, and both properties are required.
51       */
52      public RateLimitRule(long requests, TimeUnit timeUnit) {
53          this(null, requests, timeUnit);
54      }
55  
56      /**
57       * Creates a limit with an optional repository pattern, bucket capacity and refill time unit.
58       * The pattern must be {@code null} for a global limit and nonempty for a repository rule.
59       * A count of {@code -1} allows unlimited requests, and {@code 0} rejects all requests.
60       * The count must be at least {@code -1}, and both {@code requests} and {@code timeUnit} are required.
61       */
62      @JsonCreator
63      public RateLimitRule(@JsonProperty("pattern") @Nullable String pattern,
64                           @JsonProperty(value = "requests", required = true) long requests,
65                           @JsonProperty(value = "timeUnit", required = true) TimeUnit timeUnit) {
66          checkArgument(pattern == null || !pattern.isEmpty(), "pattern must not be empty");
67          this.pattern = pattern != null ? Pattern.compile(pattern) : null;
68          checkArgument(requests >= -1, "requests: %s (expected: >= -1)", requests);
69          this.requests = requests;
70          this.timeUnit = requireNonNull(timeUnit, "timeUnit");
71      }
72  
73      /**
74       * Returns the regular expression matched against {@code projectName/repoName},
75       * or {@code null} for a global limit.
76       */
77      @Nullable
78      @JsonProperty
79      @JsonInclude(Include.NON_NULL)
80      public String pattern() {
81          return pattern != null ? pattern.pattern() : null;
82      }
83  
84      /**
85       * Returns the bucket capacity and the number of tokens replenished per time unit,
86       * or {@code -1} for unlimited requests and {@code 0} to reject all requests.
87       */
88      @JsonProperty
89      public long requests() {
90          return requests;
91      }
92  
93      /**
94       * Returns the refill time unit. {@link TimeUnit#DAYS} means exactly 24 hours.
95       */
96      @JsonProperty
97      public TimeUnit timeUnit() {
98          return timeUnit;
99      }
100 
101     /**
102      * Returns whether the entire {@code projectName/repoName} matches this rule,
103      * or {@code true} if no pattern is specified.
104      */
105     public boolean matches(String projectAndRepoName) {
106         return pattern == null || pattern.matcher(projectAndRepoName).matches();
107     }
108 
109     @Override
110     public String toString() {
111         return MoreObjects.toStringHelper(this)
112                           .omitNullValues()
113                           .add("pattern", pattern())
114                           .add("requests", requests)
115                           .add("timeUnit", timeUnit)
116                           .toString();
117     }
118 }
