16  package com.linecorp.centraldogma.server;
17  
18  import static com.google.common.base.Preconditions.checkArgument;
19  
20  import java.util.List;
21  
22  import org.jspecify.annotations.Nullable;
23  
24  import com.fasterxml.jackson.annotation.JsonCreator;
25  import com.fasterxml.jackson.annotation.JsonProperty;
26  import com.google.common.base.MoreObjects;
27  import com.google.common.collect.ImmutableList;
28  
29  /**
30   * Per-server rate limits for HTTP API v1 content pushes and reverts.
31   * The first matching repository rule is checked before the shared global limit.
32   */
33  public final class RateLimitingConfig {
34  
35      static final RateLimitingConfig DISABLED = new RateLimitingConfig(null, ImmutableList.of());
36  
37      @Nullable
38      private final RateLimitRule global;
39      private final List<RateLimitRule> repositories;
40  
41      /**
42       * Creates rate limiting configuration with an optional global limit and ordered repository rules.
43       * A missing global limit or repository rule list disables the corresponding limits.
44       * The global limit must not have a pattern, and every repository rule must have a pattern.
45       */
46      @JsonCreator
47      public RateLimitingConfig(@JsonProperty("global") @Nullable RateLimitRule global,
48                                @JsonProperty("repositories") @Nullable List<RateLimitRule> repositories) {
49          checkArgument(global == null || global.pattern() == null, "global limit must not have a pattern");
50          this.global = global;
51          this.repositories = repositories != null ? ImmutableList.copyOf(repositories) : ImmutableList.of();
52          for (RateLimitRule repository : this.repositories) {
53              checkArgument(repository.pattern() != null, "repository rule must have a pattern");
54          }
55      }
56  
57      /**
58       * Returns the shared global limit, or {@code null} if disabled.
59       */
60      @Nullable
61      @JsonProperty
62      public RateLimitRule global() {
63          return global;
64      }
65  
66      /**
67       * Returns the ordered repository rules. The first full-string match wins.
68       * Each repository has an independent budget, even if it matches the same rule as another repository.
69       */
70      @JsonProperty
71      public List<RateLimitRule> repositories() {
72          return repositories;
73      }
74  
75      @Override
76      public String toString() {
77          return MoreObjects.toStringHelper(this)
78                            .add("global", global)
79                            .add("repositories", repositories)
80                            .toString();
81      }
82  }
